# Triton include directories needed by tritonUtils.cpp and
# ScheduleHintUtils.cpp. The `third_party/amd/include` paths cover the
# top-level Triton AMD headers; the `third_party` parents cover the
# `amd/include/...`-prefixed includes those headers use internally to
# pull in autogenerated TableGen output (TritonAMDGPUAttrDefs.h.inc,
# Dialect.h.inc, ...).
target_include_directories(MLIRRockUtility PRIVATE
  ${CMAKE_SOURCE_DIR}/external/triton/third_party/amd/include
  ${TRITON_BINARY_DIR}/third_party/amd/include
  ${TRITON_PROJECT_DIR}/third_party
  ${TRITON_BINARY_DIR}/third_party
  ${TRITON_PROJECT_DIR}/include
  ${TRITON_BINARY_DIR}/include
)
target_include_directories(obj.MLIRRockUtility PRIVATE
  ${CMAKE_SOURCE_DIR}/external/triton/third_party/amd/include
  ${TRITON_BINARY_DIR}/third_party/amd/include
  ${TRITON_PROJECT_DIR}/third_party
  ${TRITON_BINARY_DIR}/third_party
  ${TRITON_PROJECT_DIR}/include
  ${TRITON_BINARY_DIR}/include
)

  target_link_libraries(obj.MLIRRockUtility PUBLIC hip::host hip::amdhip64)
endif()
